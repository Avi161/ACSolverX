# PyTorch-only stack for experiments/ppo/ and tests/ppo/. Nothing there imports JAX, so
# this file is deliberately NOT a superset of requirements.txt -- installing that would
# pull the whole JAX/orbax/flax training stack for a package that never touches it.
#
# No CUDA build is pinned on purpose: the training runs use the torch that ships
# preinstalled on the Colab / vast.ai GPU image, and reinstalling it there breaks CUDA
# wheel matching (see experiments/ppo/VAST_AI.md and the SETUP cell of
# experiments/notebooks/ppo/ppo_baseline.ipynb). This file exists so CI and a laptop can
# install a CPU build and run the suite. Note that on Linux, PyPI's default torch wheel
# is the CUDA build, and pip prefers it even with --extra-index-url -- to force the CPU
# wheel, install torch alone with the index pinned:
#
#   pip install numpy==2.1.3
#   pip install "torch>=2.2" --index-url https://download.pytorch.org/whl/cpu
#
# Floor is 2.2: everything used is long-stable, and every torch.load call already passes
# weights_only=False explicitly, so torch >= 2.6's default flip is a non-issue.
torch>=2.2
numpy==2.1.3
# Not used by experiments/ppo/ itself -- the two shaping tests import the heuristic
# reference (experiments/search/heuristics.py), which sits on the numba'd solver.
numba==0.63.1
